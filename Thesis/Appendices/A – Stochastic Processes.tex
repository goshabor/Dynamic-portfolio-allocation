\section{Doob's Sampling Theorem}
\label{Appendix: Doob's Sampling Theorem}
\begin{theorem}[{\cite[Theorem 3.6c]{BhattacharyaWaymire2007}} Optional Stopping]
    Let $\{X(t)\}_{t \in [0,T]}$ be a righ-continuous $\{\F_t\}$-martingale, where $T \in [0,\infty)$. Suppose $\tau$ is a stopping time such that 
    \begin{enumerate}[label=(\roman*)]
        \item 
        \begin{equation}
            \Prob(\tau < \infty) = 1,
            \label{eqn: Optional Sampling Theorem (cond. i)}
        \end{equation}
        and
        \item 
        \begin{equation}
            X(\tau \wedge t) \quad \text{is uniformly integrable for } t \in [0,T].
            \label{eqn: Optional Sampling Theorem (cond. i)}
        \end{equation}
        Then,
        \begin{equation*}
            \E[X(\tau)] = \E[X(0)].
        \end{equation*}
    \end{enumerate}
\end{theorem}

\section{Chebyshev's Inequality}
\label{Appendix: Chebyshev's Inequality}
\begin{theorem}[{\cite[Chapter 2]{wainwright2019high}} Chebyshev's Inequality]
    Let $X$ be an integrable random variable with finite, nonzero variance $\sigma^2$ and mean $\mu$. Then, for any real number $k>0$, 
    \begin{equation*}
        \Prob(|X-\mu| \geq k) \leq \frac{\text{Var}[X]}{k^2}. 
    \end{equation*}
\end{theorem}

\section{Markov's Inequality}
\label{Appendix: Markov's Inequality}
\begin{theorem}[{\cite[Chapter 2]{wainwright2019high}} Markov's Inequality]
    Let $X$ be an integrable random variable with finite, nonzero $k$-th order moment. Then, for any real number $k$ and $t>0$,
    \begin{equation*}
        \Prob(|X|^k \leq t) \leq \frac{\E[|X|^k]}{t^k}.
    \end{equation*}
\end{theorem}

\section{Fatou's Lemma}
\label{Appendix: Fatou's Lemma}
\begin{lemma}[{\cite[Fatou's Lemma]{dibenedetto2002real}}] 
    Let $\{X, \mathcal A, \mu \}$ be a measure space and $E$ a measurable set. Let $\{f_n\}$ be a sequence of measurable functions and a.e. nonnegative function in $E$. Then,
    \begin{equation*}
        \int_E \liminf_{n \rightarrow \infty} f_n \diff \mu \leq \liminf_{n \rightarrow \infty} \int_E f_n \diff \mu.
    \end{equation*}
\end{lemma}
